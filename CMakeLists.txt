cmake_minimum_required(VERSION 3.24)
project(ExecutorLab LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(luau GIT_REPOSITORY https://github.com/luau-lang/luau.git GIT_TAG 0.747 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(luau)
add_library(ExecutorCore SHARED src/ExecutorCore.cpp)
target_link_libraries(ExecutorCore PRIVATE Luau.Compiler Luau.VM)
add_executable(TestHost src/TestHost.cpp)
add_executable(ExecutorGUI WIN32 src/ExecutorGUI.cpp)
target_link_libraries(ExecutorGUI PRIVATE user32 gdi32 comdlg32)
add_executable(CoreTests tests/CoreTests.cpp)
target_link_libraries(CoreTests PRIVATE ExecutorCore)
foreach(t ExecutorCore TestHost ExecutorGUI CoreTests)
  set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
  target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /EHsc)
  endif()
endforeach()
enable_testing()
add_test(NAME CoreTests COMMAND $<TARGET_FILE:CoreTests>)
set_tests_properties(CoreTests PROPERTIES TIMEOUT 25)
